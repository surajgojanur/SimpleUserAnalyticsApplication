  2. Decisions that refine the earlier architecture
        No npm workspaces. Each app has its own package.json, which keeps Render and
        Vercel deployment simple.




                                                                                                     Page 1 of 34
InsightFlow Analytics: Complete Build Guide




        The tracker is bundled with esbuild into apps/api/public/tracker.js and
        committed. Express serves it at /sdk/tracker.js , so no separate CDN or cross-folder
        build is needed on Render.
        Ingestion CORS is open (no credentials). The Origin check happens inside the handler,
        because a CORS preflight cannot know the site_id. Dashboard routes are restricted to
        DASHBOARD_ORIGIN .
        The tracker posts text/plain using fetch with keepalive: true . This avoids a
        preflight request and still survives page navigation.
        Inserts are idempotent: a retried event_id returns success instead of an error.
        A GET /api/sites endpoint feeds the dashboard's site selector.
        A seed script creates the demo site and synthetic sessions, so an evaluator sees data
        immediately.
        Read APIs are unauthenticated because the data is synthetic. State this as a trade-off
        in the README.

       res.status(404).json({ error: 'Not found' });
     }


     export function errorHandler(err, req, res, next) {
       if (err instanceof ZodError) {
         return res.status(400).json({
           error: 'Validation failed',
           details: err.issues.map((i) => ({ path: i.path.join('.'), message:
     i.message })),
         });
       }
       if (err.status) return res.status(err.status).json({ error: err.message });
       console.error(err);
       res.status(500).json({ error: 'Internal server error' });
     }

     const countIf = (type) => ({ $sum: { $cond: [{ $eq: ['$event_type', type] },
     1, 0] } });


     export async function listSessions(siteId, page, limit) {
         const [sessions, totalRow] = await Promise.all([
           Event.aggregate([
                 { $match: { site_id: siteId } },
                 {
                      $group: {
                        _id: '$session_id',
                          event_count: { $sum: 1 },
                          page_views: countIf('page_view'),
                          clicks: countIf('click'),
                          started_at: { $min: '$timestamp' },
                        last_seen_at: { $max: '$timestamp' },
                      },
                 },
                 { $sort: { last_seen_at: -1 } },
                 { $skip: (page - 1) * limit },
                 { $limit: limit },
                 {
                      $project: {
                          _id: 0,
                          session_id: '$_id',
                          event_count: 1,
                          page_views: 1,
                          clicks: 1,
                          started_at: 1,
                        last_seen_at: 1,
                      },
               },
             ]),
             Event.aggregate([
               { $match: { site_id: siteId } },
                 { $group: { _id: '$session_id' } },
                 { $count: 'total' },
           ]),
         ]);
         return { sessions, page, limit, total: totalRow[0]?.total ?? 0 };
     }

     export function getSessionEvents(siteId, sessionId) {
         return Event.find({ site_id: siteId, session_id: sessionId })
           .sort({ timestamp: 1, _id: 1 })
             .select('-_id event_type page_url timestamp click.x click.y')
             .lean();
     }


     export function listPages(siteId) {
       return Event.aggregate([
             { $match: { site_id: siteId } },
             { $group: { _id: '$page_url', views: countIf('page_view'), clicks:
     countIf('click') } },
         { $sort: { clicks: -1 } },
           { $project: { _id: 0, page_url: '$_id', views: 1, clicks: 1 } },
         ]);
     }


     export async function getHeatmap(siteId, pageUrl, limit = 5000) {
       const filter = { site_id: siteId, page_url: pageUrl, event_type: 'click' };
         const [total, rows] = await Promise.all([
           Event.countDocuments(filter),
         Event.find(filter).sort({ timestamp: -1 }).limit(limit).select('-_id
     click').lean(),
         ]);
         return { page_url: pageUrl, total_clicks: total, clicks: rows.map((r) =>
     r.click) };
     }

       const [state, setState] = useState({ loading: true, data: null, error: null
     });
          useEffect(() => {
             let live = true;
             setState((s) => ({ ...s, loading: true, error: null }));

                 .then((data) => live && setState({ loading: false, data, error: null
     }))

     e.message }));
             return () => {
                 live = false;
             };
             // eslint-disable-next-line react-hooks/exhaustive-deps
          }, deps);

     import { useEffect, useRef } from 'react';


     const W = 800;
     const median = (arr) => [...arr].sort((a, b) => a - b)[Math.floor(arr.length
     / 2)];


     export default function ClickHeatmap({ clicks, mode }) {
       const ref = useRef(null);
       const ratio = median(clicks.map((c) => c.document_height)) /
     median(clicks.map((c) => c.document_width));
         const H = Math.min(4000, Math.round(W * ratio));

         useEffect(() => {
             const ctx = ref.current.getContext('2d');
             ctx.clearRect(0, 0, W, H);
             ctx.fillStyle = '#fff';
             ctx.fillRect(0, 0, W, H);
             ctx.strokeStyle = '#e2e8f0';
         for (let i = 1; i < 10; i++) {
           ctx.beginPath(); ctx.moveTo((W * i) / 10, 0); ctx.lineTo((W * i) / 10,
     H); ctx.stroke();
           ctx.beginPath(); ctx.moveTo(0, (H * i) / 10); ctx.lineTo(W, (H * i) /
     10); ctx.stroke();
         }
             for (const c of clicks) {
               const x = (c.x / c.document_width) * W;
                 const y = (c.y / c.document_height) * H;
                 if (mode === 'dots') {
                   ctx.fillStyle = 'rgba(220,38,38,0.7)';
                   ctx.beginPath(); ctx.arc(x, y, 5, 0, Math.PI * 2); ctx.fill();
                 } else {
                   const g = ctx.createRadialGradient(x, y, 0, x, y, 28);
                      g.addColorStop(0, 'rgba(239,68,68,0.35)');
                      g.addColorStop(1, 'rgba(239,68,68,0)');
                      ctx.fillStyle = g;
                      ctx.fillRect(x - 28, y - 28, 56, 56);
                 }
           }
         }, [clicks, mode, H]);

  Assumptions and trade-offs to state honestly:

        A session is one browser on one site with at most 30 minutes of inactivity, stored in
        localStorage (with an in-memory fallback). This is a design choice, not part of the
        assignment.
        Page URLs are stored as origin plus pathname. Query strings and fragments are
        dropped to avoid capturing sensitive data, which means pages that differ only by
        query string are merged.
        The site ID is public. The Origin check stops casual misuse from other websites, but it
        is not authentication, because non-browser clients can forge headers.
        Read endpoints are unauthenticated because the demo uses synthetic data. A real
        deployment would add authentication before collecting visitor data.
        Events are sent one by one with fetch and keepalive , with no retry queue. This
        keeps the tracker small but can lose events when the network or API is down.
        The heatmap scales clicks by document size. Different responsive layouts can move
        content, so it is approximate, and desktop and mobile can be filtered separately.
        Sessions are computed from events with MongoDB aggregation instead of a separate
        collection. This is simple and consistent, but it will need a materialised sessions
        collection at large scale.
        No consent banner or bot filtering is included.
